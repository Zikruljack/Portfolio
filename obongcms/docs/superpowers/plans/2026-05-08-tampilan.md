# Tampilan Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implementasi modul Tampilan di CI4 Redaksi (pengaturan visual + widget layout) dan integrasi Go Edge untuk baca `setting.json` saat SSR.

**Architecture:** CI4 menyimpan pengaturan di `cms_tampilan` dan widget layout di `cms_widget_layout`, lalu merge keduanya menjadi `setting.json` dan push ke Go Edge via `GoEdge::publishContent()`. Go Edge membaca `setting.json` per domain saat render dan inject data setting + widget yang sudah di-render ke template.

**Tech Stack:** CI4 PHP 8.4, Twig 3, MySQL, Go + Chi v5, html/template

> **Dependency:** Task 8–10 (Go Edge) harus dikerjakan SETELAH plan `2026-05-08-theme-system-go-edge.md` selesai dieksekusi (karena Task 8–10 mengintegrasikan ke `PageData` struct yang dibuat di plan itu).

    public function pengaturan()
    {
        Permission::can('redaksi.tampilan.read');

        $tenantId = (int) tenant_id();
        $model    = new ValPengaturanTampilan();
        $row      = $model->where('tenant_id', $tenantId)->first();

    public function simpanPengaturan()
    {
        Permission::can('redaksi.tampilan.upsert');

        $tenantId = (int) tenant_id();
        $db       = Database::connect();

        // Ambil settings lama agar field yang tidak dikirim tidak terhapus
        $existing = $db->table('cms_tampilan')->where('tenant_id', $tenantId)->get()->getRow();
        $settings = $existing ? json_decode($existing->settings, true) : [];

        // Update dari POST
        $fields = ['nama_org', 'tagline', 'footer_copyright'];
        foreach ($fields as $f) {
            $val = $this->request->getPost($f);
            if ($val !== null) {
                $settings[$f] = trim($val);
            }
        }

        // Nested: kontak
        foreach (['alamat', 'telp', 'email'] as $f) {
            $val = $this->request->getPost('kontak_' . $f);
            if ($val !== null) {
                $settings['kontak'][$f] = trim($val);
            }
        }

        // Nested: sosmed
        foreach (['facebook', 'instagram', 'youtube', 'twitter'] as $f) {
            $val = $this->request->getPost('sosmed_' . $f);
            if ($val !== null) {
                $settings['sosmed'][$f] = trim($val);
            }
        }

        // Nested: tracking
        foreach (['google_tag_manager', 'google_analytics', 'facebook_pixel', 'google_site_verify', 'custom_head'] as $f) {
            $val = $this->request->getPost('tracking_' . $f);
            if ($val !== null) {
                $settings['tracking'][$f] = $f === 'custom_head' ? $val : trim($val);
            }
        }

        // Upload logo
        $logo = $this->request->getFile('logo');
        if ($logo && $logo->isValid() && ! $logo->hasMoved()) {
            $domain  = $this->getDomainForTenant($tenantId);
            $goEdge  = new GoEdge();
            $newPath = $domain ? $goEdge->uploadMedia($domain, 'tampilan', $logo) : null;
            if ($newPath) {
                $settings['logo'] = $newPath;
            }
        }

        // Upload favicon
        $favicon = $this->request->getFile('favicon');
        if ($favicon && $favicon->isValid() && ! $favicon->hasMoved()) {
            $domain  = $this->getDomainForTenant($tenantId);
            $goEdge  = new GoEdge();
            $newPath = $domain ? $goEdge->uploadMedia($domain, 'tampilan', $favicon) : null;
            if ($newPath) {
                $settings['favicon'] = $newPath;
            }
        }

        $now = date('Y-m-d H:i:s');
        if ($existing) {
            $db->table('cms_tampilan')->update(
                ['settings' => json_encode($settings, JSON_UNESCAPED_UNICODE), 'updated_at' => $now, 'updated_by' => anggota('username')],
                ['tenant_id' => $tenantId]
            );
        } else {
            $db->table('cms_tampilan')->insert([

                'settings'   => json_encode($settings, JSON_UNESCAPED_UNICODE),
                'updated_at' => $now,
                'updated_by' => anggota('username'),
            ]);

        $this->generateDanPush($tenantId);

        return jsonSuccess('Pengaturan tampilan berhasil disimpan');
    }

    public function widget()
    {
        Permission::can('redaksi.tampilan.read');

        $tenantId = (int) tenant_id();
        $layouts  = (new ValTampilan())->where('tenant_id', $tenantId)->findAll();

        // Susun menjadi map: layout_name => widgets array
        $layoutMap = [];
        foreach ($layouts as $row) {
            $layoutMap[$row->layout_name] = json_decode($row->widgets, true) ?: [];
        }

        return twig()->render('widget', [
            'layout_map'        => $layoutMap,
            'widget_positions'  => self::WIDGET_POSITIONS,
            'widget_definitions'=> self::WIDGET_DEFINITIONS,
        ]);
    }

    public function simpanWidget()
    {
        Permission::can('redaksi.tampilan.upsert');

        $tenantId = (int) tenant_id();
        $body     = $this->request->getJSON(true);

        if (! is_array($body)) {
            return jsonErrors(['layouts' => 'Format data tidak valid']);
        }

        // Validasi widget_name harus ada di WIDGET_DEFINITIONS
        $validNames = array_column(self::WIDGET_DEFINITIONS, 'name');

        $db = Database::connect();
        $db->table('cms_widget_layout')->where('tenant_id', $tenantId)->delete();

        foreach ($body as $layoutName => $widgets) {
            if (! array_key_exists($layoutName, self::WIDGET_POSITIONS)) {
                continue; // skip posisi tidak dikenal
            }

            $cleaned = [];
            foreach ($widgets as $w) {
                if (! isset($w['widget_name']) || ! in_array($w['widget_name'], $validNames, true)) {
                    continue;
                }
                $cleaned[] = [
                    'widget_name'   => $w['widget_name'],
                    'widget_option' => $w['widget_option'] ?? [],
                ];
            }

            if (empty($cleaned)) {
                continue;
            }

            $db->table('cms_widget_layout')->insert([
                'tenant_id'   => $tenantId,
                'layout_name' => $layoutName,
                'widgets'     => json_encode($cleaned, JSON_UNESCAPED_UNICODE),
            ]);
        }

        $this->generateDanPush($tenantId);

        return jsonSuccess('Widget layout berhasil disimpan');
    }

    private function generateDanPush(int $tenantId): void
    {
        $domain = $this->getDomainForTenant($tenantId);
        if (! $domain) {
            return;
        }

        $db = Database::connect();

        // 1. Baca settings
        $row      = $db->table('cms_tampilan')->where('tenant_id', $tenantId)->get()->getRow();
        $settings = $row ? json_decode($row->settings, true) : [];

        // 2. Baca widget layouts
        $layoutRows = $db->table('cms_widget_layout')->where('tenant_id', $tenantId)->get()->getResult();
        $layouts    = [];
        foreach ($layoutRows as $lr) {
            $layouts[] = [
                'layout_name' => $lr->layout_name,
                'widgets'     => json_decode($lr->widgets, true) ?: [],
            ];
        }

        // 3. Merge
        $settingJson              = $settings;
        $settingJson['widget_layouts'] = $layouts;

        // 4. Push ke Go Edge
        $goEdge = new GoEdge();
        $goEdge->publishContent($domain, [
            ['path' => 'setting.json', 'content' => json_encode($settingJson, JSON_UNESCAPED_UNICODE | JSON_PRETTY_PRINT)],
        ]);
    }

    private function getDomainForTenant(int $tenantId): ?string
    {
        $row = Database::connect()
            ->table('app_domains')
            ->select('host')
            ->where('tenant_id', $tenantId)
            ->where('deleted_at IS NULL')
            ->where('status', 'aktif')
            ->get()->getRow();

        return $row ? $row->host : null;
    }
}

package setting

import (
	"encoding/json"
	"os"
	"path/filepath"
)

type TenantSetting struct {
	NamaOrg         string        `json:"nama_org"`
	Tagline         string        `json:"tagline"`
	Logo            string        `json:"logo"`
	Favicon         string        `json:"favicon"`
	FooterCopyright string        `json:"footer_copyright"`
	Kontak          KontakInfo    `json:"kontak"`
	Sosmed          SosmedLinks   `json:"sosmed"`
	Tracking        TrackingInfo  `json:"tracking"`
	WidgetLayouts   []WidgetLayout `json:"widget_layouts"`
}

type KontakInfo struct {
	Alamat string `json:"alamat"`
	Telp   string `json:"telp"`
	Email  string `json:"email"`
}

type SosmedLinks struct {
	Facebook  string `json:"facebook"`
	Instagram string `json:"instagram"`
	Youtube   string `json:"youtube"`
	Twitter   string `json:"twitter"`
}

type TrackingInfo struct {
	GoogleTagManager string `json:"google_tag_manager"`
	GoogleAnalytics  string `json:"google_analytics"`
	FacebookPixel    string `json:"facebook_pixel"`
	GoogleSiteVerify string `json:"google_site_verify"`
	CustomHead       string `json:"custom_head"`
}

type WidgetLayout struct {
	LayoutName string   `json:"layout_name"`
	Widgets    []Widget `json:"widgets"`
}

type Widget struct {
	WidgetName   string                 `json:"widget_name"`
	WidgetOption map[string]interface{} `json:"widget_option"`
}

// Load membaca setting.json dari folder tenant. Jika tidak ada, return default kosong.
func Load(runtimePath, domain string) TenantSetting {
	path := filepath.Join(runtimePath, "public", "tenants", domain, "setting.json")
	data, err := os.ReadFile(path)
	if err != nil {
		return TenantSetting{}
	}
	var s TenantSetting
	if err := json.Unmarshal(data, &s); err != nil {
		return TenantSetting{}
	}
	return s
}

// LayoutWidgets mengembalikan slice Widget untuk posisi tertentu.
// Return slice kosong jika posisi tidak ada.
func (s *TenantSetting) LayoutWidgets(position string) []Widget {
	for _, l := range s.WidgetLayouts {
		if l.LayoutName == position {
			return l.Widgets
		}
	}
	return nil
}

	HTML  template.HTML

// RenderAll me-render semua widget untuk posisi tertentu.
func RenderAll(runtimePath, domain, position string, s setting.TenantSetting) []RenderedWidget {
	widgets := s.LayoutWidgets(position)
	result  := make([]RenderedWidget, 0, len(widgets))
	for _, w := range widgets {
		html := render(runtimePath, domain, w, s)
		result = append(result, RenderedWidget{Name: w.WidgetName, HTML: html})
	}
	return result
}

func render(runtimePath, domain string, w setting.Widget, s setting.TenantSetting) template.HTML {
	opt := w.WidgetOption
	switch w.WidgetName {
	case "berita_terkini":
		return renderBeritaTerkini(runtimePath, domain, opt)

	}
	return ""

func optStr(opt map[string]interface{}, key, def string) string {
	if v, ok := opt[key]; ok {
		if s, ok := v.(string); ok && s != "" {
			return s
		}
	}
	return def
}

func optInt(opt map[string]interface{}, key, def int) int {
	if v, ok := opt[key]; ok {
		switch n := v.(type) {
		case float64:
			return int(n)
		case int:
			return n
		}
	}
	return def
}

func readJSON(runtimePath, domain, path string, v interface{}) error {
	full := filepath.Join(runtimePath, "public", "tenants", domain, path)
	data, err := os.ReadFile(full)
	if err != nil {
		return err
	}
	return json.Unmarshal(data, v)
}

func renderBeritaTerkini(runtimePath, domain string, opt map[string]interface{}) template.HTML {
	judul := optStr(opt, "judul", "Berita Terkini")
	limit := optInt(opt, "limit", 5)
	filter := optStr(opt, "kategori", "")

	var listData struct {

		items = filtered

		items = items[:limit]
	}

	html := fmt.Sprintf(`<div class="widget widget-berita-terkini"><h4 class="widget-title">%s</h4><ul class="widget-list">`, template.HTMLEscapeString(judul))

			template.HTMLEscapeString(item.Judul),

	html += `</ul></div>`
